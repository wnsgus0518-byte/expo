import React, { useState, useEffect } from 'react';
import { View, Text, TouchableOpacity, TextInput, ScrollView, Alert } from 'react-native';
import AsyncStorage from '@react-native-async-storage/async-storage';
import * as Notifications from 'expo-notifications';
import * as Device from 'expo-device';
import { Plus, Check, X, Clock, Calendar } from 'lucide-react';

// ✅ 알림 설정 (앱이 백그라운드일 때도 울리게)
Notifications.setNotificationHandler({
  handleNotification: async () => ({
    shouldShowAlert: true,
    shouldPlaySound: true,
    shouldSetBadge: false,
  }),
});

export default function RoutineTracker() {
  const [routines, setRoutines] = useState([]);
  const [showAddForm, setShowAddForm] = useState(false);
  const [newRoutine, setNewRoutine] = useState({ name: '', time: '' });

  // ✅ 앱 실행 시: 루틴 불러오기 + 알림 권한 요청
  useEffect(() => {
    loadRoutines();
    registerForPushNotificationsAsync();
  }, []);

  // ✅ 저장된 루틴 불러오기
  const loadRoutines = async () => {
    try {
      const data = await AsyncStorage.getItem('routines');
      if (data) setRoutines(JSON.parse(data));
    } catch (e) {
      console.log('불러오기 실패:', e);
    }
  };

  // ✅ 루틴 저장
  const saveRoutines = async (data) => {
    try {
      await AsyncStorage.setItem('routines', JSON.stringify(data));
    } catch (e) {
      console.log('저장 실패:', e);
    }
  };

  // ✅ 알림 권한 요청
  async function registerForPushNotificationsAsync() {
    if (Device.isDevice) {
      const { status: existingStatus } = await Notifications.getPermissionsAsync();
      let finalStatus = existingStatus;
      if (existingStatus !== 'granted') {
        const { status } = await Notifications.requestPermissionsAsync();
        finalStatus = status;
      }
      if (finalStatus !== 'granted') {
        Alert.alert('알림 권한이 허용되지 않았습니다 😢');
        return;
      }
    } else {
      Alert.alert('푸시 알림은 실제 기기에서만 작동합니다.');
    }
  }

  // ✅ 매일 반복 알림 설정
  const scheduleDailyNotification = async (routineName, time) => {
    try {
      const [hour, minute] = time.split(':').map(Number);

      await Notifications.scheduleNotificationAsync({
        content: {
          title: '⏰ 루틴 알림',
          body: `${routineName} 할 시간이에요!`,
          sound: true,
        },
        trigger: {
          hour,
          minute,
          repeats: true, // ✅ 매일 반복
        },
      });
    } catch (error) {
      console.log('알림 예약 실패:', error);
    }
  };

  // ✅ 루틴 완료 토글
  const toggleComplete = (id) => {
    const updated = routines.map((routine) =>
      routine.id === id
        ? { ...routine, completed: !routine.completed, streak: !routine.completed ? routine.streak + 1 : routine.streak }
        : routine
    );
    setRoutines(updated);
    saveRoutines(updated);
  };

  // ✅ 루틴 삭제
  const deleteRoutine = (id) => {
    const filtered = routines.filter((r) => r.id !== id);
    setRoutines(filtered);
    saveRoutines(filtered);
  };

  // ✅ 루틴 추가
  const addRoutine = async () => {
    if (newRoutine.name && newRoutine.time) {
      const newItem = {
        id: Date.now(),
        name: newRoutine.name,
        time: newRoutine.time,
        completed: false,
        streak: 0,
      };

      // 🟣 매일 알림 예약
      await scheduleDailyNotification(newItem.name, newItem.time);

      const updated = [...routines, newItem];
      setRoutines(updated);
      saveRoutines(updated);

      setNewRoutine({ name: '', time: '' });
      setShowAddForm(false);
    } else {
      Alert.alert('이름과 시간을 모두 입력해주세요.');
    }
  };

  const completedCount = routines.filter((r) => r.completed).length;
  const totalCount = routines.length;
  const completionRate = totalCount > 0 ? Math.round((completedCount / totalCount) * 100) : 0;

  return (
    <ScrollView style={{ flex: 1, backgroundColor: '#f9f9ff', padding: 16 }}>
      {/* Header */}
      <View style={{ backgroundColor: 'white', borderRadius: 16, padding: 20, marginBottom: 20, elevation: 2 }}>
        <View style={{ flexDirection: 'row', justifyContent: 'space-between', alignItems: 'center', marginBottom: 12 }}>
          <Text style={{ fontSize: 24, fontWeight: 'bold', color: '#333' }}>나의 루틴</Text>
          <Calendar color="#9b5de5" size={28} />
        </View>

        {/* Progress Bar */}
        <Text style={{ color: '#666', marginBottom: 8 }}>
          오늘의 진행률: {completedCount}/{totalCount} ({completionRate}%)
        </Text>
        <View style={{ height: 8, backgroundColor: '#e5e5e5', borderRadius: 8 }}>
          <View
            style={{
              height: 8,
              borderRadius: 8,
              width: `${completionRate}%`,
              backgroundColor: '#9b5de5',
            }}
          />
        </View>
      </View>

      {/* 루틴 리스트 */}
      {routines.map((routine) => (
        <View
          key={routine.id}
          style={{
            backgroundColor: 'white',
            borderRadius: 12,
            padding: 16,
            marginBottom: 10,
            borderWidth: routine.completed ? 2 : 0,
            borderColor: routine.completed ? '#4ade80' : '#fff',
          }}
        >
          <View style={{ flexDirection: 'row', alignItems: 'center', justifyContent: 'space-between' }}>
            <TouchableOpacity
              onPress={() => toggleComplete(routine.id)}
              style={{
                width: 40,
                height: 40,
                borderRadius: 20,
                backgroundColor: routine.completed ? '#4ade80' : '#e5e5e5',
                alignItems: 'center',
                justifyContent: 'center',
              }}
            >
              <Check color={routine.completed ? 'white' : '#666'} size={20} />
            </TouchableOpacity>

            <View style={{ flex: 1, marginLeft: 12 }}>
              <Text
                style={{
                  fontSize: 18,
                  fontWeight: '600',
                  textDecorationLine: routine.completed ? 'line-through' : 'none',
                  color: routine.completed ? '#999' : '#333',
                }}
              >
                {routine.name}
              </Text>
              <Text style={{ color: '#888', marginTop: 4 }}>
                <Clock size={14} /> {routine.time} | 🔥 {routine.streak}일
              </Text>
            </View>

            <TouchableOpacity onPress={() => deleteRoutine(routine.id)}>
              <X color="#f87171" size={22} />
            </TouchableOpacity>
          </View>
        </View>
      ))}

      {/* 루틴 추가 폼 */}
      {!showAddForm ? (
        <TouchableOpacity
          onPress={() => setShowAddForm(true)}
          style={{
            backgroundColor: '#9b5de5',
            padding: 16,
            borderRadius: 12,
            alignItems: 'center',
            marginTop: 10,
          }}
        >
          <Text style={{ color: 'white', fontWeight: 'bold', fontSize: 16 }}>+ 새 루틴 추가</Text>
        </TouchableOpacity>
      ) : (
        <View style={{ backgroundColor: 'white', borderRadius: 12, padding: 16, marginTop: 10 }}>
          <Text style={{ fontSize: 18, fontWeight: 'bold', color: '#333', marginBottom: 10 }}>새 루틴 추가</Text>

          <TextInput
            placeholder="루틴 이름 (예: 아침 운동)"
            value={newRoutine.name}
            onChangeText={(text) => setNewRoutine({ ...newRoutine, name: text })}
            style={{
              borderWidth: 1,
              borderColor: '#ddd',
              borderRadius: 8,
              padding: 10,
              marginBottom: 10,
            }}
          />

          <TextInput
            placeholder="시간 (예: 07:00)"
            value={newRoutine.time}
            onChangeText={(text) => setNewRoutine({ ...newRoutine, time: text })}
            style={{
              borderWidth: 1,
              borderColor: '#ddd',
              borderRadius: 8,
              padding: 10,
              marginBottom: 10,
            }}
          />

          <View style={{ flexDirection: 'row', justifyContent: 'space-between' }}>
            <TouchableOpacity
              onPress={addRoutine}
              style={{ flex: 1, backgroundColor: '#9b5de5', padding: 12, borderRadius: 8, alignItems: 'center', marginRight: 5 }}
            >
              <Text style={{ color: 'white', fontWeight: 'bold' }}>추가</Text>
            </TouchableOpacity>

            <TouchableOpacity
              onPress={() => setShowAddForm(false)}
              style={{ flex: 1, backgroundColor: '#ddd', padding: 12, borderRadius: 8, alignItems: 'center', marginLeft: 5 }}
            >
              <Text style={{ color: '#333', fontWeight: 'bold' }}>취소</Text>
            </TouchableOpacity>
          </View>
        </View>
      )}
    </ScrollView>
  );
}
